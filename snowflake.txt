 Build Node.js + Express backend to connect Snowflake and expose API

Create a folder for your backend (outside or inside your project — up to you). 
For example /backend.

Initialize a new Node.js project:

npm install express snowflake-sdk cors body-parser

npm install @mui/material @mui/icons-material @emotion/react @emotion/styled
npm install axios


Create a file, e.g. server.js, with content like below (adapt with your Snowflake credentials):

// server.js
const express = require('express');
const snowflake = require('snowflake-sdk');
const cors = require('cors');

const app = express();
app.use(cors());
app.use(express.json());

// configure Snowflake connection
const connection = snowflake.createConnection({
  account: '<your_account>',
  username: '<your_username>',
  password: '<your_password>',
  warehouse: '<your_warehouse>',       // optional, if needed
  database: 'MY_DATABASE',             // your database
  schema: 'MY_SCHEMA'                  // your schema
});

connection.connect((err, conn) => {
  if (err) {
    console.error('Unable to connect to Snowflake: ', err);
  } else {
    console.log('Connected to Snowflake successfully.');
  }
});

// GET: fetch all menu items
app.get('/api/menu', (req, res) => {
  connection.execute({
    sqlText: `SELECT * FROM menu_items ORDER BY id`,
    complete: (err, stmt, rows) => {
      if (err) {
        console.error(err);
        res.status(500).json({ error: err.message });
      } else {
        res.json(rows);
      }
    }
  });
});

// POST: add new menu item
app.post('/api/menu', (req, res) => {
  const { name, price, description } = req.body;
  const sql = `INSERT INTO menu_items (name, price, description) VALUES (?, ?, ?)`;
  connection.execute({
    sqlText: sql,
    binds: [name, price, description],
    complete: (err, stmt, rows) => {
      if (err) {
        console.error(err);
        res.status(500).json({ error: err.message });
      } else {
        res.json({ success: true });
      }
    }
  });
});

// (Optional) PUT / DELETE endpoints would be similar

const PORT = process.env.PORT || 5000;
app.listen(PORT, () => {
  console.log(`Server listening on port ${PORT}`);
});

start backend--
node server.js
Add new React page/component to fetch and display data

Inside your existing React project, create a new component, e.g. src/pages/MenuFromDB.js:


// src/pages/MenuFromDB.js
import React, { useEffect, useState } from 'react';

function MenuFromDB() {
  const [menu, setMenu] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    fetch('http://localhost:5000/api/menu')
      .then(res => {
        if (!res.ok) throw new Error('Network response was not ok');
        return res.json();
      })
      .then(data => {
        setMenu(data);
        setLoading(false);
      })
      .catch(err => {
        console.error('Fetch error:', err);
        setError(err);
        setLoading(false);
      });
  }, []);

  if (loading) return <div>Loading menu...</div>;
  if (error) return <div>Error: {error.message}</div>;

  return (
    <div>
      <h1>Cafe Menu (from Snowflake)</h1>
      <ul>
        {menu.map(item => (
          <li key={item.ID /* maybe use item.ID or item.ID1 depending on returned field name */}>
            <strong>{item.NAME || item.name}</strong> — ${item.PRICE || item.price}
            <p>{item.DESCRIPTION || item.description}</p>
          </li>
        ))}
      </ul>
    </div>
  );
}

export default MenuFromDB;

?? 
Important Security & Best Practice Notes

Do not store credentials (username/password) in front-end React code —
 always do DB access on backend.

Use environment variables (.env or secret storage) to store Snowflake credentials; 
don’t commit them to version control.

Validate and sanitize inputs in backend before executing SQL (to avoid SQL injection)
.

For production, consider more robust setup: connection pooling, error handling, and maybe an ORM or query builder.


For large data / production usage, think about pagination, caching, rate limits, etc.
